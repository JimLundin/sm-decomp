# What a clean-room export holds (MODEL §5, "The clean room is enforced by construction").
# export.sh copies [all] and [domain <name>] from the source commit, writes [generated] itself,
# Nothing else enters the export.
# `{domain}` is the domain's name. An entry ending in `/` is a directory.
# A change here widens what an implementer sees: the adversary reviews it like a brief.

[all]
# The clean crates (crates/README.md has the rest of the workspace). They depend on each other,
# std and the registry crates of allowed_deps.txt. A crate absent from the source commit is
# created empty.
crates/clean_types/
crates/clean_model/{domain}/
# The domain's spec and vectors, from the observer.
docs/behaviours/{domain}.md
scenarios/vectors/{domain}.toml
# The implementer's process and the design it builds to, written for the clean room.
.agents/skills/implementer/SKILL.md
docs/cleanroom/DESIGN.md
# Toolchain and lint configuration. The workspace lints are copied into the generated Cargo.toml.
rust-toolchain.toml
.gitattributes
# The registry crates whose build scripts are baked in (tools/vendor_nobuild.sh), as the root
# manifest's [patch.crates-io] names them; export.sh drops those the export doesn't use.
vendor/

# Entries for one domain only, as `[domain samus-motion]`.

[generated]

Cargo.toml

Cargo.lock
.cargo/config.toml
vendor/crates-io/
CLEANROOM.md

clippy.toml
